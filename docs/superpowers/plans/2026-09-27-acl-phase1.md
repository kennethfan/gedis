# ACL 一期 Implementation Plan（命令级：用户+AUTH+ACL命令+aclfile）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 用户体系与命令级鉴权可用：AUTH 登录、ACL 用户管理、命令 allow/deny、aclfile 持久化，无配置时行为与今日一致。

**Architecture:** 新增 `internal/acl` 包（用户表+规则求值+aclfile 编解码）；`network` 加 `ContextWithUser`（仿 `ContextWithConn`）；`Router.Dispatch` 内 intercept 之后加鉴权门；连接→用户映射仿 `TxnRegistry` 的 conn-keyed 会话 + `ConnClosed` 清理；`AUTH`/`ACL` 注册为普通命令。

**Tech Stack:** Go（module github.com/kennethfan/gedis），标准库 + testify（accept 测试沿用 `cluster_accept_test.go` 的 dial 帮助函数风格），TOML 用现有 `pelletier/go-toml`（config.go 已用）。

**Spec:** `docs/superpowers/specs/2026-09-27-acl-design.md`（§1、§3、§4 部分，§5 PR1 范围）

## Global Constraints

- 零用户配置时现有 `go test ./...` 零修改通过，RESP 措辞以真机 7.2.6 录制为准。
- 错误措辞逐字节：`NOAUTH Authentication required.` / `NOPERM this user has no permissions to run the '<cmd>' command` / `WRONGPASS invalid username-password pair or user is disabled.`（末尾句号）。
- 密码比对必须 `crypto/subtle.ConstantTimeCompare`，禁止直接 `==`。
- commit 步骤需用户明确批准后才可执行（先写代码与测试，commit 待批）。

## Review Focus

- `AUTH user pass` 与 `AUTH pass`（default）双形态混淆：单参数时必须按 default 用户校验，返回合理人预期（错密给 WRONGPASS 而非 arity 错）。
- `SETUSER` 规则顺序语义：`+@all -GET` 与 `-GET +@all` 结果相反，测试必须覆盖顺序敏感。
- 连接断开后重连复用 fd：conn→user 映射必须按 `net.Conn` 身份清理，不能按远端地址，否则新连接继承旧身份。
- `ACL SAVE` 无 aclfile 配置时必须回 `ERR No ACL file configured` 而非静默 OK。
- 并发：`Dispatch` 热路径读用户表用 RWMutex 读锁，`SETUSER` 写锁；压测无数据竞争（`go test -race`）。

```go
package acl

import (
	"crypto/sha256"
	"crypto/subtle"
	"fmt"
	"strings"
	"sync"
)

const DefaultUser = "default"

type ruleOp struct {
	allow bool
	token string // "@all" / "@string" / "GET" ...
}

type User struct {
	On       bool
	NoPass   bool
	passHash [32]byte
	hasPass  bool
	rules    []ruleOp
}

type Store struct {
	mu    sync.RWMutex
	users map[string]*User
}

func NewStore() *Store {
	st := &Store{users: make(map[string]*User)}
	st.users[DefaultUser] = &User{On: true, NoPass: true, rules: []ruleOp{{allow: true, token: "@all"}}}
	return st
}

func (s *Store) SetUser(name string, rules ...string) error {
	u := &User{}
	for _, r := range rules {
		switch {
		case r == "on":
			u.On = true
		case r == "off":
			u.On = false
		case r == "nopass":
			u.NoPass = true
		case strings.HasPrefix(r, ">"):
			u.passHash = sha256.Sum256([]byte(r[1:]))
			u.hasPass = true
		case strings.HasPrefix(r, "+"), strings.HasPrefix(r, "-"):
			u.rules = append(u.rules, ruleOp{allow: r[0] == '+', token: r[1:]})
		default:
			return fmt.Errorf("acl: unknown rule %q", r)
		}
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	s.users[name] = u
	return nil
}

func (s *Store) Authenticate(name, pass string) bool {
	s.mu.RLock()
	u, ok := s.users[name]
	s.mu.RUnlock()
	if !ok || !u.On {
		return false
	}
	if u.NoPass {
		return true
	}
	if !u.hasPass {
		return false
	}
	sum := sha256.Sum256([]byte(pass))
	return subtle.ConstantTimeCompare(sum[:], u.passHash[:]) == 1
}

var cmdCategory = map[string]string{
	"GET": "string", "SET": "string", "DEL": "generic", "PING": "connection",
	"AUTH": "connection", "ACL": "connection",
}

func inCategory(cmd, cat string) bool {
	if cat == "all" {
		return true
	}
	c, ok := cmdCategory[cmd]
	return ok && c == cat
}

// CanRun 按规则顺序叠加求值：命中最后一条匹配规则；无命中默认拒绝（default 用户 +@all 兜底）。
func (s *Store) CanRun(name, cmd string) bool {
	s.mu.RLock()
	u, ok := s.users[name]
	rules := append([]ruleOp(nil), u.rules...)
	s.mu.RUnlock()
	if !ok || !u.On {
		return false
	}
	allowed := false
	for _, r := range rules {
		tok := strings.ToUpper(r.token)
		match := false
		if strings.HasPrefix(tok, "@") {
			match = inCategory(strings.ToUpper(cmd), strings.TrimPrefix(strings.ToLower(tok), "@"))
		} else {
			match = strings.EqualFold(tok, cmd)
		}
		if match {
			allowed = r.allow
		}
	}
	return allowed
}

func (s *Store) UserExists(name string) bool {
	s.mu.RLock()
	defer s.mu.RUnlock()
	_, ok := s.users[name]
	return ok
}
```

```go
	s.mu.RLock()
	defer s.mu.RUnlock()
	if len(s.users) != 1 {
```

```go
package commands

import (
	"context"
	"net"
	"sync"

	"github.com/kennethfan/gedis/internal/acl"
	"github.com/kennethfan/gedis/internal/network"
	"github.com/kennethfan/gedis/internal/protocol"
)

// AuthRegistry 维护 conn→已认证用户名；断开由 ConnClosed 清理（仿 TxnRegistry 会话模式）。
type AuthRegistry struct {
	mu    sync.Mutex
	users map[net.Conn]string
	store *acl.Store
}

func RegisterAuth(r *network.Router, st *acl.Store) *AuthRegistry {
	reg := &AuthRegistry{users: make(map[net.Conn]string), store: st}
	r.Register("AUTH", reg.auth)
	return reg
}

func (reg *AuthRegistry) auth(ctx context.Context, args []protocol.Value) protocol.Value {
	var name, pass string
	switch len(args) {
	case 1:
		s, ok := argString(args[0])
		if !ok {
			return errValueStr("ERR invalid password")
		}
		name, pass = acl.DefaultUser, s
	case 2:
		u, ok := argString(args[0])
		if !ok {
			return errValueStr("ERR invalid username")
		}
		p, ok := argString(args[1])
		if !ok {
			return errValueStr("ERR invalid password")
		}
		name, pass = u, p
	default:
		return errValueStr("ERR wrong number of arguments for 'auth' command")
	}
	if !reg.store.Authenticate(name, pass) {
		return errValueStr("WRONGPASS invalid username-password pair or user is disabled.")
	}
	if conn, ok := network.ConnFromContext(ctx); ok {
		reg.mu.Lock()
		reg.users[conn] = name
		reg.mu.Unlock()
	}
	return protocol.Value{Kind: protocol.KindSimpleString, S: "OK"}
}

func (reg *AuthRegistry) ConnClosed(conn net.Conn) {
	reg.mu.Lock()
	defer reg.mu.Unlock()
	delete(reg.users, conn)
}

// UserOf 供 server.handle 查询连接当前用户；无记录返回 default。
func (reg *AuthRegistry) UserOf(conn net.Conn) string {
	reg.mu.Lock()
	defer reg.mu.Unlock()
	if u, ok := reg.users[conn]; ok {
		return u
	}
	return acl.DefaultUser
}
```

server.handle 改动：`ctx` 模板构造改为每次循环前 `ContextWithUser(base, authReg.UserOf(conn))`——server 需持有 `*commands.AuthRegistry` 会造成 network→commands 依赖（反向！）。改为接口：server 定义 `type UserProvider interface{ UserOf(net.Conn) string }` + `SetUserProvider`，handle 内 `if up != nil { ctx = ContextWithUser(ctx, up.UserOf(conn)) }`。main.go 把 `RegisterAuth` 返回的 registry 传给主/哨兵两个 server。

config：`Config` struct 加 `ACLFile string \`toml:"aclfile"\``（缺席即 ""，纯内存模式）。
